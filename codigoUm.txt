import sqlite3

# 1 EXTRAÇÃO #tentando
print ("carregando dados...")

# 2. LIMPEZA CALENDARIO
print ("Limpando Calendario...")

# Limapndo Preço
df_calendar['price'] = df_calendar['price'].astype (str).str.replace(r'[$,]', '', regex=True).astype(float)
print (df_calendar.info())
print (df_calendar.head())

# convertendo para data real
df_calendar['date'] = pd.to_datetime(df_calendar['date'])
print (df_calendar.info())

# limpeza sazonalidade
df_calendar = df_calendar.columns.str.strip()


print(type(df_calendar))
print(df_calendar.columns if hasattr(df_calendar, "columns") else "Não é DataFrame")


# transformar t/f em 1/0
df_calendar['available'] = (df_calendar['available'] == 't').astype(int)

# criar colunas de tempo
df_calendar['month'] = df_calendar['date'].dt.month
df_calendar['day'] = df_calendar['date'].dt.day
df_calendar['weekday'] = df_calendar['date'].dt.weekday
